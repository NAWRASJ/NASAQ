cmake_minimum_required(VERSION 3.24)
project(CAM VERSION 0.3.0 LANGUAGES CXX)
# Strawberry's strict legacy Qt compile definitions do not apply to this module.
set_directory_properties(PROPERTIES COMPILE_DEFINITIONS "")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Widgets Multimedia Sql Network TextToSpeech PrintSupport Test)
add_library(cam_core src/store.cpp src/scheduler.cpp)
target_include_directories(cam_core PUBLIC src)
target_link_libraries(cam_core PUBLIC Qt6::Core Qt6::Sql)
find_package(OpenSSL REQUIRED)
set(NASAQ_KEYGEN_ACCOUNT "" CACHE STRING "Public Keygen account UUID")
set(NASAQ_KEYGEN_PRODUCT "" CACHE STRING "Public Keygen product UUID")
set(NASAQ_KEYGEN_PUBLIC_KEY "" CACHE STRING "Public Ed25519 key (64 hex digits)")
foreach(field NASAQ_KEYGEN_ACCOUNT NASAQ_KEYGEN_PRODUCT NASAQ_KEYGEN_PUBLIC_KEY)
  if(NOT "${${field}}" MATCHES "^[A-Za-z0-9-]*$")
    message(FATAL_ERROR "Invalid public licensing configuration")
  endif()
endforeach()
configure_file(src/licenseconfig.h.in licenseconfig.h @ONLY)
add_library(cam_license src/licenselease.cpp src/licenseactivation.cpp)
target_include_directories(cam_license PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(cam_license PUBLIC cam_core Qt6::Network OpenSSL::Crypto)
if(WIN32)
  target_link_libraries(cam_license PRIVATE crypt32)
endif()
add_library(cam_ui src/audio.cpp src/musictransport.cpp src/window.cpp)
target_link_libraries(cam_ui PUBLIC cam_license Qt6::Widgets Qt6::Multimedia Qt6::Network Qt6::TextToSpeech Qt6::PrintSupport)
add_executable(CAM WIN32 src/main.cpp)
target_link_libraries(CAM PRIVATE cam_ui)
enable_testing()
add_executable(cam_tests tests/core_test.cpp)
target_link_libraries(cam_tests PRIVATE cam_core Qt6::Test)
add_test(NAME cam_core COMMAND cam_tests)
add_executable(cam_audio_tests tests/audio_test.cpp)
target_link_libraries(cam_audio_tests PRIVATE cam_ui Qt6::Test)
add_test(NAME cam_audio COMMAND cam_audio_tests)
install(TARGETS CAM RUNTIME DESTINATION .)

qt_add_resources(cam_ui nasaq_brand PREFIX "/nasaq" BASE assets FILES assets/nasaq-128.png)

add_executable(cam_license_tests tests/license_test.cpp)
target_link_libraries(cam_license_tests PRIVATE cam_license Qt6::Test)
add_test(NAME cam_license COMMAND cam_license_tests)
