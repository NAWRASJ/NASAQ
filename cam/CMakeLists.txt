cmake_minimum_required(VERSION 3.24)
include(${CMAKE_CURRENT_LIST_DIR}/../cmake/NasaqVersion.cmake)
project(CAM VERSION ${NASAQ_VERSION} LANGUAGES CXX)
# Strawberry's strict legacy Qt compile definitions do not apply to this module.
set_directory_properties(PROPERTIES COMPILE_DEFINITIONS "")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Widgets Multimedia Sql Network TextToSpeech PrintSupport Test)
add_library(cam_core src/store.cpp src/scheduler.cpp)
target_compile_definitions(cam_core PUBLIC NASAQ_VERSION_STRING="${NASAQ_VERSION}")
if(NASAQ_STORE_BUILD)
  target_compile_definitions(cam_core PUBLIC NASAQ_STORE_BUILD)
endif()
target_include_directories(cam_core PUBLIC src)
target_link_libraries(cam_core PUBLIC Qt6::Core Qt6::Sql)
find_package(OpenSSL REQUIRED)
set(NASAQ_KEYGEN_ACCOUNT "" CACHE STRING "Public Keygen account UUID")
set(NASAQ_KEYGEN_PRODUCT "" CACHE STRING "Public Keygen product UUID")
set(NASAQ_KEYGEN_PUBLIC_KEY "" CACHE STRING "Public Ed25519 key (64 hex digits)")
foreach(field NASAQ_KEYGEN_ACCOUNT NASAQ_KEYGEN_PRODUCT NASAQ_KEYGEN_PUBLIC_KEY)
  if(NOT "${${field}}" MATCHES "^[A-Za-z0-9-]*$")
    message(FATAL_ERROR "Invalid public licensing configuration")
  endif()
endforeach()
option(NASAQ_REQUIRE_ACTIVATION_CONFIG "Require public publisher configuration for release" OFF)
if(NASAQ_REQUIRE_ACTIVATION_CONFIG AND (NASAQ_KEYGEN_ACCOUNT STREQUAL "" OR NASAQ_KEYGEN_PRODUCT STREQUAL "" OR NASAQ_KEYGEN_PUBLIC_KEY STREQUAL ""))
 message(FATAL_ERROR "NASAQ release requires public Keygen configuration")
endif()
configure_file(src/licenseconfig.h.in licenseconfig.h @ONLY)
add_library(cam_license src/licenselease.cpp src/licenseactivation.cpp)
target_include_directories(cam_license PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(cam_license PUBLIC cam_core Qt6::Network OpenSSL::Crypto)
if(WIN32)
  target_link_libraries(cam_license PRIVATE crypt32)
endif()
add_library(cam_ui src/audience.cpp src/storeupdates.cpp src/windowsstartup.cpp src/chimes.cpp src/audio.cpp src/musictransport.cpp src/window.cpp src/workspacedialogs.cpp src/prayers.cpp src/cloudspeech.cpp src/cloudspeechdialog.cpp src/studio.cpp src/studiodialog.cpp)
find_package(Qt6 REQUIRED COMPONENTS Concurrent)
target_link_libraries(cam_ui PRIVATE Qt6::Concurrent)
if(MSVC)
  target_link_libraries(cam_ui PRIVATE windowsapp)
endif()
target_link_libraries(cam_ui PUBLIC cam_license Qt6::Widgets Qt6::Multimedia Qt6::Network Qt6::TextToSpeech Qt6::PrintSupport)
add_executable(CAM WIN32 src/main.cpp)
target_link_libraries(CAM PRIVATE cam_ui)
enable_testing()
add_executable(cam_tests tests/core_test.cpp)
target_link_libraries(cam_tests PRIVATE cam_core Qt6::Test)
add_test(NAME cam_core COMMAND cam_tests)
add_executable(cam_audio_tests tests/audio_test.cpp)
target_link_libraries(cam_audio_tests PRIVATE cam_ui Qt6::Test)
add_test(NAME cam_audio COMMAND cam_audio_tests)
install(TARGETS CAM RUNTIME DESTINATION .)

qt_add_resources(cam_ui nasaq_brand PREFIX "/nasaq" BASE assets FILES assets/adhan-default.png assets/turntable-base.png assets/turntable-record.png assets/turntable-arm.png assets/nasaq-128.png assets/nasaq.qss assets/control-up.png assets/control-up-disabled.png assets/control-down.png assets/control-down-disabled.png)

add_executable(cam_license_tests tests/license_test.cpp)
target_link_libraries(cam_license_tests PRIVATE cam_ui Qt6::Test)
add_test(NAME cam_license COMMAND cam_license_tests)
add_executable(cam_review_tests tests/review_test.cpp)
target_link_libraries(cam_review_tests PRIVATE cam_ui Qt6::Test)
add_test(NAME cam_review COMMAND cam_review_tests)
add_executable(cam_cloud_tests tests/cloud_test.cpp)
target_link_libraries(cam_cloud_tests PRIVATE cam_ui Qt6::Test)
add_test(NAME cam_cloud COMMAND cam_cloud_tests)

add_executable(cam_studio_tests tests/studio_test.cpp)
target_link_libraries(cam_studio_tests PRIVATE cam_ui Qt6::Test)
add_test(NAME cam_studio COMMAND cam_studio_tests)

add_executable(cam_business_tests tests/business_test.cpp)
target_link_libraries(cam_business_tests PRIVATE cam_ui Qt6::Test)
add_test(NAME cam_business COMMAND cam_business_tests)

add_executable(nasaq_msix_runtime_probe tests/msix_runtime_probe.cpp)
target_link_libraries(nasaq_msix_runtime_probe PRIVATE cam_core Qt6::Widgets Qt6::Gui Qt6::Sql)

add_executable(nasaq_silent_notifications_tests tests/silent_notifications_test.cpp)
target_include_directories(nasaq_silent_notifications_tests PRIVATE src)
target_link_libraries(nasaq_silent_notifications_tests PRIVATE Qt6::Widgets Qt6::Test)
add_test(NAME nasaq_silent_notifications COMMAND nasaq_silent_notifications_tests)
